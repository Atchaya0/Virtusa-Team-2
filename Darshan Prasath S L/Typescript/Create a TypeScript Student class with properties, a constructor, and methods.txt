class Student {
    
    public name: string;
    public age: number;
    private grades: number[];

    
    constructor(name: string, age: number, grades: number[] = []) {
        this.name = name;
        this.age = age;
        this.grades = grades;
    }

   
    public addGrade(grade: number): void {
        this.grades.push(grade);
    }

    public getAverageGrade(): number {
        if (this.grades.length === 0) return 0;
        const total = this.grades.reduce((sum, current) => sum + current, 0);
        return total / this.grades.length;
    }

    public getSummary(): string {
        return `${this.name} (Age: ${this.age}) - Average Grade: ${this.getAverageGrade().toFixed(2)}`;
    }
}
